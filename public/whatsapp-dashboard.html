if (typeof window !== 'undefined') { window.switchTab = function switchTab(tabName) { document.querySelectorAll('.section').forEach((section) => { section.classList.toggle('active', section.id === tabName); }); document.querySelectorAll('.tab-btn').forEach((btn) => { const onclick = btn.getAttribute('onclick') || ''; btn.classList.toggle('active', onclick.includes(`'${tabName}'`)); }); }; }
